# HTML Visualize — Type Templates (类型骨架模板)

> Each template below is a **minimum viable skeleton**. Weak models: copy and fill content. Strong models: extend freely but keep the design system CSS intact.

> [!IMPORTANT]
> **Common elements in ALL templates:**
> - `design-system.css` content inlined in `<style>`
> - Theme toggle button
> - Clipboard-safe copy function with textarea fallback
> - `<meta name="artifact-type" content="...">` declaring the type

## Shared JavaScript (所有模板共用)

Paste this `<script>` block at the bottom of every generated HTML file:

```html
<script>
// --- Theme Toggle ---
function toggleTheme() {
  const root = document.documentElement;
  const current = root.getAttribute('data-theme');
  const next = current === 'dark' ? 'light' : (current === 'light' ? 'dark' : 
    (window.matchMedia('(prefers-color-scheme: light)').matches ? 'dark' : 'light'));
  root.setAttribute('data-theme', next);
  localStorage.setItem('theme', next);
  document.querySelector('.theme-toggle').textContent = next === 'dark' ? '☀️' : '🌙';
}
(function initTheme() {
  const saved = localStorage.getItem('theme');
  if (saved) {
    document.documentElement.setAttribute('data-theme', saved);
    const btn = document.querySelector('.theme-toggle');
    if (btn) btn.textContent = saved === 'dark' ? '☀️' : '🌙';
  }
})();

// --- Clipboard Copy with file:// Fallback ---
function copyToClipboard(text, btn) {
  const originalText = btn ? btn.textContent : '';
  function onSuccess() { if (btn) { btn.textContent = '✅ Copied!'; setTimeout(() => btn.textContent = originalText, 2000); } }
  function onFail() { 
    // Fallback: textarea select method (works on file://)
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.style.cssText = 'position:fixed;left:-9999px;top:-9999px';
    document.body.appendChild(ta);
    ta.focus(); ta.select();
    try { document.execCommand('copy'); onSuccess(); } catch(e) { alert('Copy failed. Please select and copy manually:\n\n' + text.substring(0, 200) + '...'); }
    document.body.removeChild(ta);
  }
  if (navigator.clipboard && window.isSecureContext) {
    navigator.clipboard.writeText(text).then(onSuccess).catch(onFail);
  } else { onFail(); }
}

// --- Tab Switching ---
function switchTab(containerId, tabName) {
  const container = document.getElementById(containerId);
  if (!container) return;
  container.querySelectorAll('.tab-button').forEach(b => b.classList.toggle('active', b.dataset.tab === tabName));
  container.querySelectorAll('.tab-panel').forEach(p => p.classList.toggle('active', p.dataset.tab === tabName));
}

// --- Auto TOC Generation ---
function generateTOC() {
  const toc = document.querySelector('.toc ul, .toc ol');
  if (!toc) return;
  document.querySelectorAll('main h2, main h3').forEach((h, i) => {
    if (!h.id) h.id = 'section-' + i;
    const li = document.createElement('li');
    li.style.paddingLeft = h.tagName === 'H3' ? '1rem' : '0';
    li.innerHTML = '<a href="#' + h.id + '">' + h.textContent + '</a>';
    toc.appendChild(li);
  });
}
document.addEventListener('DOMContentLoaded', generateTOC);
</script>
```

### Presentation Variant (演示/Slideshow 变体)

Replace `<main>` with slide sections. Add slide navigation.

```html
<main>
  <section class="slide">
    <h2>Slide 1: Title</h2>
    <p>Key point</p>
  </section>
  <section class="slide">
    <h2>Slide 2: Content</h2>
    <div class="card-grid">
      <div class="card"><h3>Point A</h3><p>Detail</p></div>
      <div class="card"><h3>Point B</h3><p>Detail</p></div>
    </div>
  </section>
  <section class="slide">
    <h2>Slide 3: Conclusion</h2>
    <p>Summary</p>
  </section>
</main>
<nav class="slide-nav">
  <button class="btn" onclick="scrollPrev()">← Prev</button>
  <button class="btn" onclick="scrollNext()">Next →</button>
</nav>
<script>
const slides = document.querySelectorAll('.slide');
let current = 0;
function scrollTo(i) { slides[Math.max(0, Math.min(i, slides.length-1))].scrollIntoView({behavior:'smooth'}); current = i; }
function scrollPrev() { scrollTo(current - 1); }
function scrollNext() { scrollTo(current + 1); }
</script>
```

## 🎛️ Type 5: Playground (交互式调参)

Use for: Parameter tuning, algorithm visualization, A/B comparison with live preview.

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <meta name="artifact-type" content="playground">
  <title>{{TITLE}} — Playground</title>
  <style>
    /* PASTE design-system.css HERE */
    .playground-layout { display: grid; grid-template-columns: 300px 1fr; gap: 1.5rem; }
    .controls-panel { position: sticky; top: 1rem; }
    .preview-panel { min-height: 400px; }
    @media (max-width: 768px) { .playground-layout { grid-template-columns: 1fr; } .controls-panel { position: static; } }
  </style>
</head>
<body>
  <button class="theme-toggle" onclick="toggleTheme()" aria-label="Toggle theme">🌙</button>
  <div class="container-wide">
    <header>
      <h1>{{TITLE}}</h1>
      <p class="meta">Adjust parameters and see results in real time</p>
    </header>
    <main>
      <div class="playground-layout">
        <div class="controls-panel card">
          <h3>Controls</h3>
          <div class="slider-control">
            <label for="param-a">Parameter A</label>
            <input type="range" id="param-a" min="0" max="100" value="50" oninput="update()">
            <span class="slider-value" id="val-a">50</span>
          </div>
          <div class="slider-control">
            <label for="param-b">Parameter B</label>
            <input type="range" id="param-b" min="0" max="100" value="30" oninput="update()">
            <span class="slider-value" id="val-b">30</span>
          </div>
          <div style="margin-top: 1rem;">
            <button class="export-button" onclick="exportConfig(this)">📤 Copy Config</button>
          </div>
        </div>
        <div class="preview-panel card">
          <h3>Preview</h3>
          <div id="preview-area">
            <!-- Dynamic preview content -->
          </div>
        </div>
      </div>
    </main>
  </div>
  <!-- PASTE shared JS HERE -->
  <script>
  function update() {
    const a = document.getElementById('param-a').value;
    const b = document.getElementById('param-b').value;
    document.getElementById('val-a').textContent = a;
    document.getElementById('val-b').textContent = b;
    // Update preview based on parameters
    document.getElementById('preview-area').innerHTML = 
      '<p>Parameter A: ' + a + ', Parameter B: ' + b + '</p>';
  }
  function exportConfig(btn) {
    const config = JSON.stringify({
      paramA: +document.getElementById('param-a').value,
      paramB: +document.getElementById('param-b').value
    }, null, 2);
    copyToClipboard(config, btn);
  }
  update();
  </script>
</body>
</html>
```

## ✏️ Type 6: Editor (自定义编辑器)

Use for: One-time editing tools, drag-sort interfaces, config editors, prompt tuners. **MUST have export/copy button**.

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <meta name="artifact-type" content="editor">
  <title>{{TITLE}} — Editor</title>
  <style>
    /* PASTE design-system.css HERE */
    .editor-layout { display: grid; grid-template-columns: 1fr 1fr; gap: 1.5rem; }
    @media (max-width: 768px) { .editor-layout { grid-template-columns: 1fr; } }
    .editor-panel textarea {
      width: 100%; min-height: 300px; resize: vertical;
      background: var(--bg-elevated); color: var(--text-primary);
      border: 1px solid var(--border); border-radius: var(--radius);
      padding: 1rem; font-family: var(--font-mono); font-size: 0.85rem;
      line-height: 1.6;
    }
    .editor-panel textarea:focus { outline: none; border-color: var(--accent); }
    .output-panel { background: var(--bg-surface); border: 1px solid var(--border); border-radius: var(--radius-lg); padding: 1.25rem; }
  </style>
</head>
<body>
  <button class="theme-toggle" onclick="toggleTheme()" aria-label="Toggle theme">🌙</button>
  <div class="container-wide">
    <header>
      <h1>{{TITLE}}</h1>
      <p class="meta">Edit on the left, see results on the right</p>
    </header>
    <main>
      <div class="editor-layout">
        <div class="editor-panel">
          <h3>Input</h3>
          <textarea id="editor-input" oninput="onEditorChange()" placeholder="Enter content here..."></textarea>
        </div>
        <div>
          <h3>Output</h3>
          <div class="output-panel" id="editor-output">
            <!-- Rendered output -->
          </div>
          <div style="margin-top: 1rem; display: flex; gap: 0.5rem;">
            <button class="export-button" onclick="exportAsJSON(this)">📤 Copy as JSON</button>
            <button class="export-button" onclick="exportAsMarkdown(this)">📋 Copy as Markdown</button>
          </div>
        </div>
      </div>
    </main>
  </div>
  <!-- PASTE shared JS HERE -->
  <script>
  function onEditorChange() {
    const input = document.getElementById('editor-input').value;
    // Transform input → output (customize per use case)
    const escaped = input.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
    document.getElementById('editor-output').innerHTML = '<pre>' + escaped + '</pre>';
  }
  function exportAsJSON(btn) {
    const data = JSON.stringify({ content: document.getElementById('editor-input').value }, null, 2);
    copyToClipboard(data, btn);
  }
  function exportAsMarkdown(btn) {
    copyToClipboard(document.getElementById('editor-input').value, btn);
  }
  </script>
</body>
</html>
```

## Per-Type Checklist (类型特定检查项)

| Type | Extra Checks Beyond Global Gate |
|------|-------------------------------|
| **Document** | TOC auto-generates · Copy button works · Presentation variant has slide nav |
| **Diagram** | SVG renders · Colors adapt to both themes · Legend/notes present |
| **Prototype** | Mockup frames visible · Side-by-side works on mobile (stacks) |
| **Dashboard** | Metric values readable · Table sorts if applicable · Badges use correct colors |
| **Playground** | Sliders update preview in real time · Export button copies valid config/JSON |
| **Editor** | Input→output transform works · Both export buttons produce valid output · textarea focus has visual indicator |
